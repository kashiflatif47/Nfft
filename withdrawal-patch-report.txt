  247:   approve:boolean,
  248:   note='',
  249:   meta?:{ip?:string;userAgent?:string}
  250: ){

  252:     const w=await Withdrawal.findOne({
  253:       _id:id,
  254:       status:'pending'
  255:     }).session(session);
  256:

  276:     w.processedAt=new Date();

  278: 
  279:     if(!approve){
  280:       /*
  281:        * Reservation is released.
  282:        * No second balance credit is required because the original
  283:        * reservation never removed the member's actual balance.
  284:        */
  285:       reservation.status='reversed';
  286:       reservation.adminActorId=adminId;
  287:       await reservation.save({session});
  288: 
  289:       await ledgerEntry({
  290:         userId:w.userId,
  291:         type:'withdrawal',
  292:         amount:w.requestedAmount,
  293:         direction:'credit',

  296:         relatedEntity:w._id,
  297:         adminActorId:adminId
  298:       },{session});
  299: 
  300:       w.status='rejected';
  301:       await w.save({session});
  302:     }else{
  303:       /*
  304:        * APPROVAL:
  305:        * Do NOT call debitAvailable().
  306:        *
  307:        * The amount was already reserved at request time.
  308:        * Now remove the reservation and permanently deduct the
  309:        * requested amount from the member wallet exactly once.
  310:        */
  311:       const wallet=await Wallet.findOne({
  312:         userId:w.userId
  313:       }).session(session);
  314: 
  315:       if(!wallet){
  316:         throw new AppError(
  317:           404,
  318:           'Member wallet not found'
  319:         );
  320:       }
  321: 
  322:       if(wallet.withdrawalReserved<w.requestedAmount){
  323:         throw new AppError(
  324:           409,
  325:           'Withdrawal reservation is invalid'
  326:         );
  327:       }
  328: 
  329:       /*
  330:        * Convert reservation into final withdrawal.
  331:        * We remove the reserved amount and deduct from the
  332:        * actual balance fields in one transaction.
  333:        */
  334:       /*
  335:        * Withdrawals are PROFIT-WALLET ONLY.
  336:        * Locked capital, deposit, commission and rewards must never
  337:        * be used to fund a withdrawal.
  338:        */
  339:       const fields=['profit','commission','rewards','capitalAvailable'];
  340: 
  341:       let remaining=money(w.requestedAmount);
  342: 
  343:       for(const field of fields){
  344:         const current=Number(
  345:           (wallet as any)[field]??0
  346:         );
  347: 
  348:         const take=Math.min(
  349:           current,
  350:           remaining
  351:         );
  352: 
  353:         (wallet as any)[field]=money(
  354:           current-take
  355:         );
  356: 
  357:         remaining=money(
  358:           remaining-take
  359:         );
  360: 
  361:         if(remaining<=0){
  362:           break;
  363:         }
  364:       }
  365:

  367:         throw new AppError(
  368:           422,
  369:           'Insufficient balance at withdrawal approval'
  370:         );
  371:       }
  372: 
  373:       wallet.withdrawalReserved=money(
  374:         Math.max(
  375:           0,
  376:           wallet.withdrawalReserved-w.requestedAmount
  377:         )
  378:       );
  379: 
  380:       wallet.withdrawn=money(
  381:         Number(wallet.withdrawn??0)+w.requestedAmount
  382:       );
  383: 
  384:       wallet.version+=1;
  385: 
  386:       await wallet.save({session});

  388:       reservation.status='completed';
  389:       reservation.adminActorId=adminId;

  402:       }
